#include <inttypes.h>
#include "esp_log.h"
#include "rc522.h"

#define PIN_MISO 19
#define PIN_MOSI 23
#define PIN_SCLK 18
#define PIN_CS    5

static const char *TAG = "RFID";
static rc522_handle_t scanner;

static void on_tag_scanned(void *arg, esp_event_base_t base,
                           int32_t id, void *data)
{
    rc522_event_data_t *ev = (rc522_event_data_t *)data;
    if (id == RC522_EVENT_TAG_SCANNED) {
        rc522_tag_t *tag = (rc522_tag_t *)ev->ptr;
        ESP_LOGI(TAG, "UID: %" PRIu64, tag->serial_number);
    }
}

void app_main(void)
{
    rc522_config_t cfg = {
        .spi.host      = SPI3_HOST,
        .spi.miso_gpio = PIN_MISO,
        .spi.mosi_gpio = PIN_MOSI,
        .spi.sck_gpio  = PIN_SCLK,
        .spi.sda_gpio  = PIN_CS,
    };

    ESP_ERROR_CHECK(rc522_create(&cfg, &scanner));
    ESP_ERROR_CHECK(rc522_register_events(scanner,
                    RC522_EVENT_ANY, on_tag_scanned, NULL));
    ESP_ERROR_CHECK(rc522_start(scanner));
    ESP_LOGI(TAG, "Acerque una tarjeta RFID...");
}
